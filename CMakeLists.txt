cmake_minimum_required(VERSION 3.22)
project(FluteShift VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FLUTESHIFT_BUILD_PLUGIN "Build JUCE VST3 and standalone" ON)
enable_testing()
add_executable(dsp_tests tests/dsp_tests.cpp)
target_include_directories(dsp_tests PRIVATE Source)
add_test(NAME dsp_tests COMMAND dsp_tests)
if(FLUTESHIFT_BUILD_PLUGIN)
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 51a8a6d7aeae7326956d747737ccf1575e61e209)
  FetchContent_MakeAvailable(JUCE)
  juce_add_plugin(FluteShift COMPANY_NAME "FluteShift" BUNDLE_ID "audio.fluteshift.live"
    PLUGIN_MANUFACTURER_CODE Flsh PLUGIN_CODE Fs01 FORMATS VST3 Standalone
    PRODUCT_NAME "FluteShift" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
    COPY_PLUGIN_AFTER_BUILD FALSE)
  target_sources(FluteShift PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
  target_compile_definitions(FluteShift PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
  target_link_libraries(FluteShift PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
  add_executable(plugin_tests tests/plugin_tests.cpp)
  target_include_directories(plugin_tests PRIVATE Source "${juce_SOURCE_DIR}/modules")
  target_compile_definitions(plugin_tests PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
  target_link_libraries(plugin_tests PRIVATE FluteShift)
  add_test(NAME plugin_tests COMMAND plugin_tests)
endif()
